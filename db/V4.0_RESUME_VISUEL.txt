┌──────────────────────────────────────────────────────────────────────────┐
│  1. VALIDATION PASSWORD HASHÉ                           [CRITIQUE]        │
├──────────────────────────────────────────────────────────────────────────┤
│  ❌ AVANT : Passwords peuvent être stockés en clair                      │
│  ✅ APRÈS : Contrainte DB force bcrypt/argon2                            │
│                                                                           │
│  SQL:                                                                     │
│    CONSTRAINT check_password_hashed CHECK (                              │
│      password REGEXP '^\\$2[aby]\\$[0-9]{2}\\$.{53}$'  -- bcrypt        │
│      OR password REGEXP '^\\$argon2(id|i|d)\\$'         -- argon2       │
│    )                                                                      │
│                                                                           │
│  IMPACT : -95% risque en cas de leak DB                                  │
└──────────────────────────────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────────────────────────────┐
│  2. TOKENS HASHÉS (SHA-256)                             [CRITIQUE]        │
├──────────────────────────────────────────────────────────────────────────┤
│  ❌ AVANT : Tokens stockés en plaintext → exploitables si leak           │
│  ✅ APRÈS : Tokens hashés SHA-256 → inutilisables si leak                │
│                                                                           │
│  Tables modifiées :                                                       │
│    • email_validation_tokens  (token → token_hash)                       │
│    • password_reset_tokens    (token → token_hash)                       │
│    • validation_tokens        (token → token_hash)                       │
│                                                                           │
│  Process :                                                                │
│    Backend → Génère token → Hash SHA-256 → Stocke HASH en DB            │
│             → Envoie TOKEN ORIGINAL par email                            │
│    Validation → Hash token reçu → Compare avec DB                        │
│                                                                           │
│  IMPACT : -90% risque exploitation tokens                                │
└──────────────────────────────────────────────────────────────────────────┘

💻 EXEMPLE CODE BACKEND (Node.js)

┌─────────────────────────────────────────────────────────────────────────┐
│  PASSWORD HASHING (bcrypt)                                              │
└─────────────────────────────────────────────────────────────────────────┘

const bcrypt = require('bcrypt');

// Inscription
const hashedPassword = await bcrypt.hash(plainPassword, 12);
await db.query(
  'INSERT INTO utilisateurs (email, password, ...) VALUES (?, ?, ...)',
  [email, hashedPassword, ...]
);
// ✅ La contrainte CHECK valide le format automatiquement!

// Connexion
const user = await db.query('SELECT * FROM utilisateurs WHERE email = ?', [email]);
const isValid = await bcrypt.compare(plainPassword, user.password);

┌─────────────────────────────────────────────────────────────────────────┐
│  TOKEN HASHING (SHA-256)                                                │
└─────────────────────────────────────────────────────────────────────────┘

const crypto = require('crypto');

// Génération token
const token = crypto.randomBytes(32).toString('base64url');
const tokenHash = crypto.createHash('sha256').update(token).digest('hex');

// Stockage (HASH uniquement!)
await db.query(
  'INSERT INTO email_validation_tokens (token_hash, ...) VALUES (?, ...)',
  [tokenHash, ...]
);

// Email (TOKEN ORIGINAL)
await sendEmail(user.email, { token: token });  // ← Pas le hash!

// Validation
const receivedToken = req.params.token;
const receivedHash = crypto.createHash('sha256').update(receivedToken).digest('hex');
const dbToken = await db.query(
  'SELECT * FROM email_validation_tokens WHERE token_hash = ?',
  [receivedHash]
);
