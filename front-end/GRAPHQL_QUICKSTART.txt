export const useUtilisateurs = () => {
  return useQuery({
    queryKey: ['utilisateurs'],
    queryFn: async () => {
      const response = await fetch(apiUrl('utilisateurs'), {
        credentials: 'include',
      });
      if (!response.ok) throw new Error('Failed');
      return response.json();
    },
  });
};

  }

    users: data?.users ?? [],

  };

// 1. Définir la mutation GraphQL
export const CREATE_USER = gql`
  mutation CreateUser($input: CreateUserInput!) {
    createUser(input: $input) {
      id
      first_name
      last_name
      email
    }
  }
`;

// 2. Générer les types
// npm run codegen

// 3. Utiliser dans un hook
import { useCreateUserMutation } from '@/lib/apollo/generated/graphql';

export const useCreateUser = () => {
  const [createUser, { loading, error }] = useCreateUserMutation({
    // Mettre à jour le cache après création
    update: (cache, { data }) => {
      cache.modify({
        fields: {
          users(existing = []) {
            return [...existing, data.createUser];
          },
        },
      });
    },
  });

  return {
    createUser,
    isLoading: loading,
    error,
  };
};

// 4. Utiliser dans un composant
const CreateUserForm = () => {
  const { createUser, isLoading } = useCreateUser();

  const handleSubmit = async (formData) => {
    await createUser({
      variables: { input: formData },
    });
  };

  return <Form onSubmit={handleSubmit} />;
};

🐛 DÉPANNAGE
================================================================================

Problème : "Cannot find module '@/lib/apollo/generated/graphql'"
Solution : Exécutez npm run codegen pour générer les types

Problème : Erreurs CORS
Solution : Vérifiez que le serveur GraphQL accepte les requêtes depuis
          localhost:5173 (voir api/src/graphql-server.ts)

Problème : Types non à jour après modification du schéma
Solution :
  1. Relancez le serveur GraphQL (cd ../api && npm run dev)
  2. Regénérez les types (npm run codegen)

Problème : Cache Apollo ne se met pas à jour
Solution : Utilisez refetchQueries ou update dans les mutations
