import Redis from 'ioredis';

let redis: Redis | null = null;

export function getRedisClient(): Redis {
  if (!redis) {
    redis = new Redis(process.env.REDIS_URL || 'redis://localhost:6379', {
      password: process.env.REDIS_PASSWORD,
      db: parseInt(process.env.REDIS_DB || '0'),
      retryStrategy: (times) => {
        const delay = Math.min(times * 50, 2000);
        return delay;
      },
      maxRetriesPerRequest: 3,
    });

    redis.on('connect', () => {
      console.log('✅ [Redis] Connecté');
    });

    redis.on('error', (err) => {
      console.error('❌ [Redis] Erreur:', err);
    });
  }

  return redis;
}

export async function closeRedisConnection() {
  if (redis) {
    await redis.quit();
    redis = null;
  }
}

import { getRedisClient } from '../config/redis.js';

interface RateLimitConfig {
  requests: number;
  windowMs: number;
}

interface RateLimitResult {
  allowed: boolean;
  limit: number;
  remaining: number;
  resetAt: number;
  retryAfter?: number;
}

const RATE_LIMITS: Record<string, RateLimitConfig> = {
  ANONYMOUS: { requests: 10, windowMs: 15 * 60 * 1000 },
  FREE: { requests: 100, windowMs: 15 * 60 * 1000 },
  BASIC: { requests: 500, windowMs: 15 * 60 * 1000 },
  PREMIUM: { requests: 2000, windowMs: 15 * 60 * 1000 },
  ENTERPRISE: { requests: 10000, windowMs: 15 * 60 * 1000 },
  API: { requests: 100, windowMs: 15 * 60 * 1000 },
};

class RateLimitService {
  private redis = getRedisClient();

  async checkLimit(
    tenantId: string,
    identifier: string,
    plan: string
  ): Promise<RateLimitResult> {
    const config = RATE_LIMITS[plan] || RATE_LIMITS.BASIC;
    const key = `ratelimit:${tenantId}:${identifier}`;
    const now = Date.now();
    const windowStart = now - config.windowMs;

    // Utiliser Redis Sorted Set avec timestamp comme score
    const multi = this.redis.multi();
    
    // Supprimer les entrées hors fenêtre
    multi.zremrangebyscore(key, 0, windowStart);
    
    // Compter les requêtes dans la fenêtre
    multi.zcard(key);
    
    // Ajouter cette requête
    multi.zadd(key, now, `${now}-${Math.random()}`);
    
    // Définir expiration
    multi.expire(key, Math.ceil(config.windowMs / 1000));

    const results = await multi.exec();
    const count = (results?.[1]?.[1] as number) || 0;
    
    const allowed = count < config.requests;
    const remaining = Math.max(0, config.requests - count - 1);
    const resetAt = now + config.windowMs;
    const retryAfter = allowed ? undefined : Math.ceil(config.windowMs / 1000);

    return {
      allowed,
      limit: config.requests,
      remaining,
      resetAt,
      retryAfter,
    };
  }

  async resetLimit(tenantId: string, identifier: string): Promise<void> {
    const key = `ratelimit:${tenantId}:${identifier}`;
    await this.redis.del(key);
  }

  async getStats(tenantId: string): Promise<any> {
    const pattern = `ratelimit:${tenantId}:*`;
    const keys = await this.redis.keys(pattern);
    
    let totalRequests = 0;
    for (const key of keys) {
      const count = await this.redis.zcard(key);
      totalRequests += count;
    }

    return {
      totalRequests,
      activeUsers: keys.length,
    };
  }
}

export default new RateLimitService();

─────────────────────────────────────────────────────────────────
☁️  PRODUCTION - AWS ELASTICACHE
─────────────────────────────────────────────────────────────────

1. Créer un cluster Redis via AWS Console
   • Type : cache.t3.micro (pour démarrer)
   • Engine : Redis 7.x
   • Node type : 1 node
   • Coût : ~$12/mois

2. Configurer le Security Group
   • Autoriser port 6379 depuis votre backend EC2/Lambda

3. Obtenir l'endpoint
   your-redis-cluster.abc123.cache.amazonaws.com:6379

4. Configurer l'application
   REDIS_URL=redis://your-redis-cluster.abc123.cache.amazonaws.com:6379
   REDIS_TLS=true

🧪 TESTER REDIS
─────────────────────────────────────────────────────────────────
