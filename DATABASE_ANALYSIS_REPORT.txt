ANNEXE C: Script migration colonnes DEPRECATED
----------------------------------------

-- Phase 1: Créer une vue de compatibilité
CREATE OR REPLACE VIEW v_utilisateurs_compatible AS
SELECT
  u.*,
  us.email_verified as email_verified_new,
  up.grade_id as grade_id_new,
  usub.status_id as status_id_new
FROM utilisateurs u
LEFT JOIN user_security us ON u.id = us.user_id
LEFT JOIN user_profiles up ON u.id = up.user_id
LEFT JOIN user_subscriptions usub ON u.id = usub.user_id;

-- Phase 2: Migrer les données manquantes
UPDATE user_security us
INNER JOIN utilisateurs u ON us.user_id = u.id
SET us.email_verified = u.email_verified
WHERE us.email_verified IS NULL AND u.email_verified IS NOT NULL;

-- Phase 3: Marquer comme deprecated dans les commentaires
ALTER TABLE utilisateurs
  MODIFY COLUMN email_verified TINYINT(1)
  COMMENT 'DEPRECATED - Use user_security.email_verified';

-- Vérifier la performance des index
SELECT
  TABLE_NAME,
  INDEX_NAME,
  CARDINALITY,
  ROUND(CARDINALITY / TABLE_ROWS * 100, 2) as selectivite_pct
FROM information_schema.STATISTICS s
JOIN information_schema.TABLES t
  ON s.TABLE_SCHEMA = t.TABLE_SCHEMA
  AND s.TABLE_NAME = t.TABLE_NAME
WHERE s.TABLE_SCHEMA = 'clubmanager_test'
  AND s.INDEX_NAME LIKE 'idx_%'
  AND t.TABLE_ROWS > 0
ORDER BY selectivite_pct DESC
LIMIT 20;
