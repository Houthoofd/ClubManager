C. api/src/routes/magasin/core/resolvers/magasin.resolvers.ts
──────────────────────────────────────────────────────────────

Ligne ~415-448 (obtenirStocksArticleResolver):
REMPLACER ENTIÈREMENT PAR:

const obtenirStocksArticleResolver = async (
  _: any,
  args: { id: number },
  context: any
) => {
  try {
    console.log(`📦 [MagasinResolver] Récupération stocks article: ${args.id}`);

    const stocks = await prisma.articleStock.findMany({
      where: { article_id: args.id },
      include: {
        article: true,
        taille: true
      }
    });

    return {
      success: true,
      data: stocks.map(s => ({
        id: s.id,
        article_id: s.article_id,
        taille_id: s.taille_id,
        taille: s.taille,
        quantity_total: s.quantity_total,
        quantity_available: s.quantity_available,
        quantity_reserved: s.quantity_reserved,
        low_stock_threshold: s.low_stock_threshold,
        last_restock_at: s.last_restock_at
      })),
      message: "Stocks récupérés avec succès"
    };

  } catch (error: any) {
    console.error(`❌ [MagasinResolver] Erreur stocks:`, error);
    throw new GraphQLError(
      `Erreur lors de la récupération des stocks: ${error.message}`,
      {
        extensions: { code: "INTERNAL_SERVER_ERROR", originalError: error },
      }
    );
  }
};

3. Restaurer l'ancien schema:
   mv api/prisma/schema.prisma api/prisma/schema-phase1-broken.prisma
   mv api/prisma/schema-old.prisma api/prisma/schema.prisma

4. Régénérer Prisma Client:
   cd api && npx prisma generate

5. Redémarrer l'application
